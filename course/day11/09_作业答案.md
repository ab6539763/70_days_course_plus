## 题目二参考答案

```python
def test_homework_encoding(tmp_path):
    utf8_path = tmp_path / "utf8_only.txt"
    utf8_path.write_text("智链科技", encoding="utf-8")
    content, enc = read_text_file(utf8_path)
    assert enc == "utf-8"
    assert "智链" in content

    gbk_path = tmp_path / "gbk_only.txt"
    gbk_path.write_bytes("智链科技".encode("gbk"))
    content2, enc2 = read_text_file(gbk_path)
    assert enc2 == "gbk"
```

**讲解**：`read_text_file` 先 `read_bytes` 再尝试解码；GBK 文件不会被 utf-8 误判成功（会 UnicodeDecodeError 继续下一编码）。

## 题目三参考答案

```python
from pathlib import Path
from tools.doc_reader import iter_text_files, read_document
from core.exceptions import StorageError


def count_lines_in_directory(directory: Path, pattern: str = "*.txt") -> dict[str, int]:
    if not directory.exists():
        raise StorageError(f"目录不存在: {directory}", path=str(directory))
    result: dict[str, int] = {}
    for file_path in iter_text_files(directory, pattern):
        record = read_document(file_path, clean=False)
        result[file_path.name] = len(record.content.splitlines())
    return result
```

**讲解**：复用 `iter_text_files` 保证排序与异常行为一致；行数用 `splitlines()` 比 `count('\n')` 更准确（末行无换行）。

## 题目四参考答案

```python
import pytest
from tools.doc_reader import mask_phone_numbers


@pytest.mark.parametrize(
    "text,expected_count,should_mask",
    [
        ("无号码", 0, False),
        ("13812345678", 1, True),
        ("12345678901", 0, False),  # 非大陆 11 位 1 开头规则
        ("13812345678和13900001111", 2, True),
    ],
)
def test_mask_phone_homework(text, expected_count, should_mask):
    masked, count = mask_phone_numbers(text)
    assert count == expected_count
    if should_mask:
        assert "***PHONE***" in masked
        assert "13812345678" not in masked or "13900001111" not in masked
```

**讲解**：正则 `1\d{10}` 要求以 1 开头共 11 位数字；产品级脱敏 Day 28 会升级更严格规则。
