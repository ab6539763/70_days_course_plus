## 题目三：validate_phone 扩展

```python
def validate_phone(phone: str) -> str | None:
    phone = (phone or "").strip().replace("-", "").replace(" ", "")
    if not phone:
        return "手机号不能为空"
    if not re.match(r"^1\d{10}$", phone):
        return "手机号格式不正确（需 11 位且以 1 开头）"
    return None
```

## 题目四：search_by_email_domain

```python
def search_by_email_domain(contacts: list, domain: str) -> list:
    domain = (domain or "").strip().lstrip("@").lower()
    if not domain:
        return list(contacts)
    suffix = "@" + domain
    return [
        c for c in contacts
        if c.get("email", "").lower().endswith(suffix)
    ]
```

def merge_seed(store, seed_contacts):
    existing_phones = {c["phone"] for c in store["contacts"]}
    for item in seed_contacts:
        if item["phone"] in existing_phones:
            continue
        contact = create_contact(store["next_id"], ...)
