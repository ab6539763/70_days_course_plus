```python

chunks = [f"chunk-{i}" for i in range(8)]

# 取前 3 个（可能用于摘要）
head = chunks[:3]

# 取后 2 个（最新内容）
tail = chunks[-2:]
```

## 练习块 C：删除操作对比

```python
data = ["a", "b", "c", "b", "d"]

data.remove("b")  # ['a', 'c', 'b', 'd']

x = data.pop(1)   # x='c', data=['a', 'b', 'd']

data.clear()      # []
```

## 练习块 H：模拟 todo_manager 核心逻辑

```python
todos = []
next_id = 1

def add(title, pri):
    global next_id
    todos.append([next_id, title, "", pri, False])
    next_id += 1

def complete(tid):
    for t in todos:
        if t[0] == tid:
            t[4] = True
            return True
    return False

add("学 list", 1)
add("写待办", 2)
complete(1)

pending = [t[1] for t in todos if not t[4]]
print("未完成:", pending)
```
