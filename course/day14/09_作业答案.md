"""homework test — Day 14"""

from pathlib import Path
import sys

import pytest

SRC = Path(__file__).resolve().parents[2] / "src"
sys.path.insert(0, str(SRC))

from chat.cli_assistant import ChatAssistant
from llm.client import LLMClient
from llm.env import LLMEnvConfig
from models import ModelConfig

SAMPLE = {
    "model": "deepseek-chat",
    "choices": [{"message": {"role": "assistant", "content": "ok"}, "finish_reason": "stop"}],
    "usage": {"prompt_tokens": 1, "completion_tokens": 1, "total_tokens": 2},
}


def make_client():
    env = LLMEnvConfig(api_key="t", base_url="https://api.example.com/v1", mock=True)
    return LLMClient(ModelConfig(), env=env, transport=lambda *_: SAMPLE)


def test_scripted_history_after_chat():
    a = ChatAssistant(client=make_client())
    outs = a.run_scripted(["你好", "再问", "/history"])
    assert any("--- 对话历史 ---" in o for o in outs)


def test_unknown_command():
    a = ChatAssistant(client=make_client(), system_prompt="")
    outs = a.run_scripted(["/foo"])
    assert any("未知命令" in o for o in outs)


def test_save_creates_file(tmp_path):
    path = tmp_path / "hw.json"
    a = ChatAssistant(client=make_client(), history_path=path)
    a.run_scripted(["hi", "/save"])
    assert path.exists()

## 作业 3：代码阅读问答

### 1. `__init__` 顺序

1. `self.history = history or MessageHistory()`  
2. `self.history_path = history_path or get_path(...)`  
3. `self.client = client or ResilientLLMClient(...)`  
4. 若 `system_prompt` 且尚无 system → `add_system`

先有待操作的历史对象，再绑定客户端，最后注入人设。

### 2. `handle_command` 三元组

| 字段 | 含义 |
|------|------|
| `handled` | 是否已作为命令处理（含未知命令） |
| `message` | 展示给用户的字符串 |
| `should_exit` | 是否应结束主循环 |

### 3. 为何 `chat_turn` 不捕获 APIError

- 保持 `chat_turn` 职责单一：完成一轮对话逻辑  
- `_process_line` 统一将异常转为用户可读消息，交互与脚本路径一致  
- 测试可对 `chat_turn` 单独断言异常抛出，对 `run_scripted` 断言友好输出

## 作业 6：Day 15 预习要点

`complete` 返回的 `usage` 含 `prompt_tokens`（输入）、`completion_tokens`（输出）。多轮对话每轮携带**全部历史**，`prompt_tokens` 随轮次增长，导致延迟与费用上升。Day 15 Token 计数器将累计展示，为 `/clear` 与截断策略提供数据依据。
