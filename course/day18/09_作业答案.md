```python
from __future__ import annotations

import sys
from pathlib import Path

SRC = Path(__file__).resolve().parents[2] / "nexus-agent-platform" / "src"
sys.path.insert(0, str(SRC))

from prompts.intent import (
    DEFAULT_KEYWORD_RULES,
    INTENT_TEMPLATE_MAP,
    RuleBasedIntentClassifier,
)

CUSTOM_RULES = {
    **DEFAULT_KEYWORD_RULES,
    "meeting_minutes": ["纪要", "会议记录", "议题", "决议", "晨会"],
}

CUSTOM_MAP = {
    **INTENT_TEMPLATE_MAP,
    "meeting_minutes": "doc_summary",
}

clf = RuleBasedIntentClassifier(rules=CUSTOM_RULES, template_map=CUSTOM_MAP)


def classify_meeting(text: str):
    return clf.classify(text)


if __name__ == "__main__":
    samples = ["请整理会议纪要议题", "你好"]
    for s in samples:
        m = classify_meeting(s)
        print(m.summary())
```

```python
router = IntentRouter(context_provider=lambda: "测试上下文" * 50)
```

## 作业 C 参考答案要点

`threshold_design.md` 核心：

```python
def safe_route_and_apply(router, assistant, user_text, min_conf=0.65):
    match = router.classify(user_text)
    if match.confidence < min_conf:
        return match, "请说明您需要：查文档 / 总结 / 合规审阅 / 产品咨询？"
    router.route_and_apply(assistant, user_text)
    return match, None
```

| 例句 | 是否澄清 | 理由 |
|------|----------|------|
| 你好 | 可能（0.6） | 低于 0.65 |
| 帮我看看 | 是 | 零命中 general 0.6 |
| 请总结报告三个要点 | 否 | doc_summary 命中 ≥1，confidence ≥0.65 |
| 审阅宣传语合规 | 否 | 多词命中 |

对 `test_classify_general_fallback`：`confidence=0.6 < 0.65` 将不自动路由，与现网 `auto_route` 行为不同——需在文档说明特性开关。

## 作业 D 参考答案

| 对比项 | Day 17 /template | Day 18 auto_route |
|--------|------------------|-------------------|
| 触发时机 | 用户敲命令 | 每条 user 消息前 |
| 是否调 LLM | 否（仅切换） | 是（切换后 complete） |
| 变量谁提供 | 默认 template_variables 或命令无参 | IntentRouter.build_variables |
| 适用角色 | 开发/运营调试 | 终端用户 |

1. 以为 `/route` 会切换模板——实际只预览

## 作业 E 参考答案骨架

```python
router = IntentRouter()
assistant = ChatAssistant(client=client, intent_router=router, auto_route=True)
while True:
    line = input(">>> ").strip()
    if line in ("/exit", "/quit"):
        break
    if assistant.is_command(line):
        handled, msg, exit_ = assistant.handle_command(line)
        print(msg)
        if exit_:
            break
    else:
        print(assistant.chat_turn(line))
    name = assistant.prompt_template.name if assistant.prompt_template else "none"
```
